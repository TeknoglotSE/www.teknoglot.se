{{- /* Prev/next navigation, shown in the sidebar on post pages only. Hexo's post.prev is the NEWER post and post.next the OLDER one. In Hugo the equivalent is the next/previous element of the date-sorted sequence, which is inverted relative to Hexo's naming. */ -}} {{- $all := where site.RegularPages "Type" "posts" -}} {{- $all = $all.ByDate.Reverse -}} {{- $idx := -1 -}} {{- range $i, $p := $all -}}{{ if eq $p.RelPermalink $.RelPermalink }}{{ $idx = $i }}{{ end }}{{ end -}} {{- $newer := "" -}} {{- $older := "" -}} {{- if ge $idx 0 -}} {{- if gt $idx 0 -}}{{- $newer = index $all (sub $idx 1) -}}{{- end -}} {{- if lt $idx (sub (len $all) 1) -}}{{- $older = index $all (add $idx 1) -}}{{- end -}} {{- end -}} {{- if or $newer $older -}}
{{- with $newer -}} newer

{{ .Title }}

 {{- end -}} {{- with $older -}} older
